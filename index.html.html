Skrip SQL Database Supabase

Salin dan jalankan skrip SQL berikut di Supabase SQL Editor Anda. Skrip menggunakan IF NOT EXISTS agar aman dijalankan tanpa error relasi ganda:

-- 1. Tabel Master Chart of Accounts (COA)
CREATE TABLE IF NOT EXISTS chart_of_accounts (
  account_id VARCHAR(20) PRIMARY KEY,
  account_name VARCHAR(100) NOT NULL,
  normal_balance VARCHAR(6) NOT NULL CHECK (normal_balance IN ('DEBIT', 'CREDIT')),
  report_classification VARCHAR(50) NOT NULL CHECK (report_classification IN ('ASSET', 'LIABILITY', 'EQUITY', 'REVENUE', 'EXPENSE'))
);

-- Master Data Akun Sampel

-- 2. Tabel Header Jurnal Transaksi
CREATE TABLE IF NOT EXISTS journal_headers (
  header_id SERIAL PRIMARY KEY,
  trx_date DATE NOT NULL DEFAULT CURRENT_DATE,
  ref_number VARCHAR(50) UNIQUE NOT NULL,
  description TEXT,
  posting_status VARCHAR(10) NOT NULL DEFAULT 'DRAFT' CHECK (posting_status IN ('DRAFT', 'POSTED', 'VOID')),
  created_by VARCHAR(50)
);

-- 3. Tabel Detail Rincian Jurnal
CREATE TABLE IF NOT EXISTS journal_lines (
  line_id SERIAL PRIMARY KEY,
  header_id INT NOT NULL REFERENCES journal_headers(header_id) ON DELETE CASCADE,
  account_id VARCHAR(20) NOT NULL REFERENCES chart_of_accounts(account_id) ON DELETE RESTRICT,
  debit NUMERIC(15, 2) NOT NULL DEFAULT 0.00 CHECK (debit >= 0),
  credit NUMERIC(15, 2) NOT NULL DEFAULT 0.00 CHECK (credit >= 0),
  CONSTRAINT chk_debit_credit_not_both_zero CHECK (debit > 0 OR credit > 0)
);
